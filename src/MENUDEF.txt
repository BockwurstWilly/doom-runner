// Doom Runner main menu (replaces the Doom / Brutal Doom main menu)

{
	Class "RunnerMainMenu"
	StaticPatch 60, 4, "RLOGO"
	Selector "M_SKULL1", -32, -5
	Font "BigFont", "Red"
	LineSpacing 18
	Position 100, 84
	TextItem "$RR_MNU_PLAY", "l", "PlayerclassMenu"
	TextItem "$RR_MNU_OPT", "o", "RunnerOptions"
	RunnerLangItem "$RR_MNU_LANG", "s", "None"
	TextItem "BRUTAL DOOM", "b", "BrutalMenu"
	TextItem "$RR_MNU_QUIT", "q", "QuitMenu"
}

{

	Title "$RR_OPT_TITLE"

	StaticText "$RR_OPT_GFXHINT", "DarkGray"
	RunnerGoreOption "$RR_OPT_GORE", "rr_gore", "RunnerGore"
	StaticText " "
	Slider "$RR_OPT_SFX", "snd_sfxvolume", 0, 1, 0.05, 2
	Slider "$RR_OPT_MUS", "snd_musicvolume", 0, 1, 0.05, 2

	Option "$RR_OPT_FULL", "vid_fullscreen", "OnOff"

	Submenu "$RR_OPT_KEYS", "CustomizeControls"
